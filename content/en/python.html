import FreeCAD as App
import Part
import math

# Create a new document or use the active one
doc = App.activeDocument() or App.newDocument("Wrench")

# ============================================
# EDIT THESE NUMBERS TO CHANGE THE WRENCH SIZE
# ============================================
nut_size = 5.5    # M3=5.5, M4=7, M5=8, M6=10, M8=13
thick = 3.0       # Wrench thickness (mm)
h_rad = 6.0       # Head radius (mm)
l_len = 50.0      # Handle length (mm)
l_wid = 7.0       # Handle width (mm)
fillet_radius = 0.5  # Edge smoothness (0.5=subtle, 1.5=very round)

# Create hexagon shape for the nut hole
hex_r = (nut_size / 2.0) / math.cos(math.radians(30))
hex_vertices = []
for i in range(6):
    angle = i * math.pi * 2 / 6
    x = hex_r * math.cos(angle)
    y = hex_r * math.sin(angle)
    hex_vertices.append(App.Vector(x, y, 0))

# Turn the hexagon into a 3D hole
hex_poly = Part.makePolygon(hex_vertices + [hex_vertices[0]])
hex_face = Part.Face(Part.Wire(hex_poly))
hex_ext = hex_face.extrude(App.Vector(0, 0, thick))

# Create the round head
head_cyl = Part.makeCylinder(h_rad, thick)

# Create the handle
handle_box = Part.makeBox(l_len, l_wid, thick)
handle_box.translate(App.Vector(0, -l_wid/2.0, 0))

# Combine head and handle, then cut out the hex hole
wrench_raw = head_cyl.fuse(handle_box)
wrench_with_hole = wrench_raw.cut(hex_ext)

# Round all the edges (makes it comfortable to hold)
try:
    final_shape = wrench_with_hole.makeFillet(fillet_radius, wrench_with_hole.Edges)
except:
    final_shape = wrench_with_hole

# Add the wrench to your FreeCAD document
obj = doc.addObject("Part::Feature", "My_Wrench")
obj.Shape = final_shape
doc.recompute()

# Zoom to see your new wrench
if App.GuiUp:
    Gui.SendMsgToActiveView("ViewFit")

print("✅ Done! Your wrench is ready. Go to File → Export to save as STL or OBJ.")
